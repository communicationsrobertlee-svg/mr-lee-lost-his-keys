const NUMBER_AUDIO_WORDS=["nol","satu","dua","tiga","empat","lima","enam","tujuh","delapan","sembilan","sepuluh"];
let gameAudioCtx=null,numberAudioBuffers={},numberAudioLoadPromise=null,lastNumberSource=null;
function unlockGameAudio(){
  try{
    const AC=window.AudioContext||window.webkitAudioContext;
    if(!AC)return Promise.resolve(false);
    if(!gameAudioCtx)gameAudioCtx=new AC();
    if(gameAudioCtx.state==="suspended")gameAudioCtx.resume();
    if(!numberAudioLoadPromise){
      numberAudioLoadPromise=Promise.all(["audio/numbers1.json?v=universal2","audio/numbers2.json?v=universal2"].map(async url=>{
        const r=await fetch(url);if(!r.ok)throw new Error(url);return r.json();
      })).then(async packs=>{
        const encoded=Object.assign({},...packs);
        await Promise.all(NUMBER_AUDIO_WORDS.map(async word=>{
          const b64=encoded[word];if(!b64)throw new Error(`audio ${word}`);
          const bin=atob(b64),bytes=new Uint8Array(bin.length);
          for(let i=0;i<bin.length;i++)bytes[i]=bin.charCodeAt(i);
          numberAudioBuffers[word]=await gameAudioCtx.decodeAudioData(bytes.buffer.slice(0));
        }));
        document.body.dataset.audioReady="true";return true;
      }).catch(()=>false);
    }
    return numberAudioLoadPromise;
  }catch(e){return Promise.resolve(false)}
}
function speechFallback(word){
  try{

    const u=new SpeechSynthesisUtterance(word);u.lang="id-ID";u.rate=.75;u.volume=1;
    window.speechSynthesis.cancel();window.speechSynthesis.speak(u);
  }catch(e){}
}
function playNumberAudio(word){
  try{
    if(!gameAudioCtx||!numberAudioBuffers[word])return false;
    if(lastNumberSource)try{lastNumberSource.stop()}catch(e){}
    const src=gameAudioCtx.createBufferSource();
    const gain=gameAudioCtx.createGain();gain.gain.value=1.0;
    src.buffer=numberAudioBuffers[word];src.connect(gain);gain.connect(gameAudioCtx.destination);
    src.start(0);lastNumberSource=src;return true;
  }catch(e){return false}
}
function speakIndonesianNumber(word){
  const clean=String(word||"").toLowerCase();
  if(playNumberAudio(clean))return;
  unlockGameAudio().then(ok=>{if(!ok||!playNumberAudio(clean))speechFallback(clean)});
}

function makePlatforms(stage){
  platforms=[];
  if(stage.gaps===0)platforms.push({x:0,y:GROUND,w:worldWidth,h:100,kind:"ground"});
  else{
    let cursor=0;const usable=worldWidth-1500,spacing=usable/stage.gaps;
    for(let i=0;i<stage.gaps;i++){
      const gx=850+i*spacing+(i%2)*80,gw=70+level*9+(i%3)*16;
      if(gx>cursor)platforms.push({x:cursor,y:GROUND,w:gx-cursor,h:100,kind:"ground"});
      cursor=gx+gw;
    }
    platforms.push({x:cursor,y:GROUND,w:worldWidth-cursor,h:100,kind:"ground"});
  }
  const raised=level===1?3:4+level;
  for(let i=0;i<raised;i++){
    const x=430+i*(worldWidth-1100)/raised+(i%2)*38;

    platforms.push({x,y,w:150+(i%3)*36,h:22,kind:"platform"});
  }
}
