function updateEnemies(dt,prevY){
  const stage=STAGES[level];
  for(const e of enemies){
    if(!e.alive)continue;
    if(e.state==="deadbin"||e.state==="broccoli")continue;
    if(e.state==="bin"){
      if(time>=e.stateUntil){e.state="flex";e.stateUntil=time+.9;e.y=GROUND-e.h;e.speechText="Aku kembali!";e.speechUntil=time+1.1;say(`💪 MR LEE IS BACK!`,.5)}
      else continue;
    }

    e.speechCooldown-=dt;
    if(e.speechCooldown<=0){e.speechText=randomMrLeeShout();e.speechUntil=time+1.45;e.speechCooldown=rand(3.0,6.8)}

    if(e.state==="flex"){
      if(time>=e.stateUntil)e.state="walk";
    }else{
      e.x+=e.vx*dt;if(e.x<e.baseX-e.range||e.x>e.baseX+e.range)e.vx*=-1;
      e.fireCooldown-=dt;e.flexCooldown-=dt;
      if(e.flexCooldown<=0){e.state="flex";e.stateUntil=time+rand(.55,.9);e.flexCooldown=rand(2.8,5.2)}
      else if(e.fireCooldown<=0&&Math.abs(e.x-player.x)<820){throwFood(e);e.fireCooldown=stage.fire+rand(0,.55)}
    }

    if(rects(player,e)){
      const stomp=player.vy>0&&prevY+player.h<=e.y+18;
      if(stomp){e.state="broccoli";e.stateUntil=Infinity;player.y=e.y-player.h-3;player.vy=-430;particlesAt(e.x+e.w/2,e.y+e.h/2,28,"#58b957");say(`🥦 STOMP! MR LEE IS NOW BROCCOLI!<br><span style="font-size:.55em">HE IS STAYING LIKE THAT.</span>`,1.15)}
      else death(`Mr Lee caught you!`,false);
    }
  }
}

function updateBoss(dt,prevY){
  if(!boss||!boss.alive)return;
  boss.x+=boss.vx*dt;if(boss.x<boss.baseX-boss.range||boss.x>boss.baseX+boss.range)boss.vx*=-1;
  boss.fireCooldown-=dt;boss.flexCooldown-=dt;boss.speechCooldown-=dt;
  if(boss.speechCooldown<=0){boss.speechText=randomMrLeeShout();boss.speechUntil=time+1.5;boss.speechCooldown=rand(2.5,5.6)}
  if(boss.flexCooldown<=0){boss.flexUntil=time+.72;boss.flexCooldown=rand(2.4,4.4)}
  if(boss.fireCooldown<=0){const count=boss.shotsPerAttack;for(let i=0;i<count;i++){const spread=(i-(count-1)/2)*.17;throwFood(boss,chooseFood(),spread,boss.foodSpeed)}boss.fireCooldown=boss.fireRate}

    else death(`${boss.name} caught you!`,false);
  }
}

function update(dt){
  time+=dt;if(messageTimer>0){messageTimer-=dt;if(messageTimer<=0)messageEl.style.opacity="0"}
  const prevY=player.y;let dir=0;if(keys.ArrowLeft||keys.KeyA)dir--;if(keys.ArrowRight||keys.KeyD)dir++;
  const enamSpeed=handsOn()?1.65:1;
  const speed=(325+Math.min(level*5,40))*(player.ridingRary?1.28:1)*enamSpeed,accel=(player.ridingRary?3150:2500)*(handsOn()?1.45:1);
  if(dir){player.vx+=dir*accel*dt;player.facing=dir}else player.vx*=Math.pow(.002,dt);
  player.vx=clamp(player.vx,-speed,speed);
  if(keys.Space){player.vy-=2450*dt;player.vy=Math.max(player.vy,-540)}
  player.vy+=GRAV*dt;player.x+=player.vx*dt;player.x=clamp(player.x,0,worldWidth-player.w);player.y+=player.vy*dt;resolvePlatforms(dt);
  if(player.y>H+180)death(`You fell off the course!`,true);

  for(const p of wordPickups)if(!p.collected&&rects(player,p))collectWord(p);
  if(expected<WORDS.length){
    for(const d of decoyPickups){
      if(!d.collected&&rects(player,d)){

  }

  }

  for(const f of foods){f.x+=f.vx*dt;f.y+=f.vy*dt;f.vy+=260*dt;f.life-=dt;const fb={x:f.x-f.r,y:f.y-f.r,w:f.r*2,h:f.r*2};if(rects(fb,player)){f.life=0;hitByFood(f)}}
  foods=foods.filter(f=>f.life>0&&f.y<H+130);

  particles=particles.filter(p=>p.life>0);

  cameraX+=(clamp(player.x-W*.34,0,worldWidth-W)-cameraX)*Math.min(1,dt*6);updateHUD();
}
