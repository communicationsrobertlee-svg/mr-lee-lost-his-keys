))collectBossPower(p);
  if(boss&&boss.alive&&time>=bossPowerSpawnAt)spawnBossPowerCluster();
  // ENAM TUJUH automatic fire is handled globally so normal and boss pickups behave identically.
}

function startBossSequence(resumeIndex=0){
  enemies.forEach(e=>e.alive=false);foods=[];
  if(level===5){
    bossQueue=[()=>bossTemplate("Medium Mr Lee",50,92,54,1.65,1,10)];
    bossIndex=-1;
  }else{
    bossQueue=[
      ()=>bossTemplate("Small Mr Lee",10,58,46,1.85,1,5),
      ()=>bossTemplate("Medium Mr Lee",50,92,58,1.60,1,12),
      ()=>bossTemplate("Large Mr Lee",100,138,68,1.35,2,20),
      ()=>bossTemplate("MEGA EVOLUTION MR LEE",200,205,78,1.05,3,28)
    ];
    bossIndex=Math.max(0,Math.min(3,resumeIndex))-1;
  }
  spawnNextBoss();
}

function spawnNextBoss(){
  bossIndex++;
  if(bossIndex>=bossQueue.length){boss=null;exitDoor.open=true;say(`🏆 MR LEE DEFEATED!<br><span style="font-size:.55em">THE EXIT IS OPEN!</span>`,2);return}
  boss=bossQueue[bossIndex]();bossPowerups=[];bossPowerSpawnAt=time+.35;
  const subtitles=level===10?["10 HP — EASY.","50 HP — NOW HE'S ANGRY.","100 HP — HE'S GETTING HUGE.","200 HP — ROBOT MUSCLE MEGA EVOLUTION!"]:["50 HP — MEDIUM MR LEE!"];
  say(`💪 ${boss.name.toUpperCase()}!<br><span style="font-size:.55em">${subtitles[bossIndex]} POWER-UPS INCOMING!</span>`,2);
}

function bossHit(dmg){
  if(!boss||!boss.alive)return;
  boss.hp-=dmg;boss.flexUntil=time+.45;particlesAt(boss.x+boss.w/2,boss.y+boss.h/2,8,"#ffb64d");
  if(boss.hp<=0){
    boss.alive=false;
    bossBin={x:boss.x,y:GROUND-Math.min(135,boss.h*.55),w:Math.max(62,boss.w*.72),h:Math.min(135,boss.h*.55),until:time+1.1};
    particlesAt(boss.x+boss.w/2,boss.y+boss.h/2,60,"#ffe45d");
    say(`🗑️ ${boss.name.toUpperCase()} IS NOW A BIN!`,1.0);
    setTimeout(()=>{if(running&&!levelTransition)spawnNextBoss()},900);
  }
}

function completeLevel(){
  if(levelTransition||!hasKey||!exitDoor.open)return;
  levelTransition=true;running=false;
  if(level<10){say(`✅ LEVEL ${level} COMPLETE!<br><span style="font-size:.55em">${STAGES[level+1].name} IS NEXT.</span>`,1.6);setTimeout(()=>{initLevel(level+1);running=true;last=performance.now();requestAnimationFrame(loop)},1300)}
  else{
    overlay.style.display="flex";
    document.getElementById("card").innerHTML=`<h1>🏆 SPORTS SHED OPEN!</h1><p>You searched all over Marangaroo, collected <b>NOL → SEPULUH</b>, recovered the keys, survived the healthy-food attacks, rode Rary, and defeated the <b>ROBOT MUSCLE MEGA EVOLUTION MR LEE</b>.</p><p>Marangaroo is ready for a week of <b>Indonesian and sport</b>.</p><p><b>Total deaths:</b> ${deaths}</p><button id="againBtn">PLAY AGAIN</button>`;
    document.getElementById("againBtn").onclick=()=>{overlay.style.display="none";deaths=0;time=0;initLevel(1);running=true;last=performance.now();requestAnimationFrame(loop)};
  }
}

function resolvePlatforms(dt){
  player.onGround=false;const prevBottom=player.y+player.h-player.vy*dt;
  for(const p of platforms){if(player.x+player.w<=p.x||player.x>=p.x+p.w)continue;if(player.vy>=0&&prevBottom<=p.y+8&&player.y+player.h>=p.y){player.y=p.y-player.h;player.vy=0;player.onGround=true}}
}

function updateEnemies(dt,prevY){
  const stage=STAGES[level];
  for(const e of enemies){
    if(!e.alive)continue;
    if(e.state==="deadbin"||e.state==="broccoli")continue;
    if(e.state==="bin"){
      if(time>=e.stateUntil){e.state="flex";e.stateUntil=time+.9;e.y=GROUND-e.h;e.speechText="Aku kembali!";e.speechUntil=time+1.1;say(`💪 MR LEE IS BACK!`,.5)}
      else continue;
    }

    e.speechCooldown-=dt;
    if(e.speechCooldown<=0){e.speechText=randomMrLeeShout();e.speechUntil=time+1.45;e.speechCooldown=rand(3.0,6.8)}

    if(e.state==="flex"){
      if(time>=e.stateUntil)e.state="walk";
    }else{
      e.x+=e.vx*dt;if(e.x<e.baseX-e.range||e.x>e.baseX+e.range)e.vx*=-1;
      e.fireCooldown-=dt;e.flexCooldown-=dt;
      if(e.flexCooldown<=0){e.state="flex";e.stateUntil=time+rand(.55,.9);e.flexCooldown=rand(2.8,5.2)}
      else if(e.fireCooldown<=0&&Math.abs(e.x-player.x)<820){throwFood(e);e.fireCooldown=stage.fire+rand(0,.55)}
    }

    if(rects(player,e)){
      const stomp=player.vy>0&&prevY+player.h<=e.y+18;
      if(stomp){e.state="broccoli";e.stateUntil=Infinity;player.y=e.y-player.h-3;player.vy=-430;particlesAt(e.x+e.w/2,e.y+e.h/2,28,"#58b957");say(`🥦 STOMP! MR LEE IS NOW BROCCOLI!<br><span style="font-size:.55em">HE IS STAYING LIKE THAT.</span>`,1.15)}
      else death(`Mr Lee caught you!`,false);
    }
  }
}

function updateBoss(dt,prevY){
  if(!boss||!boss.alive)return;
  boss.x+=boss.vx*dt;if(boss.x<boss.baseX-boss.range||boss.x>boss.baseX+boss.range)boss.vx*=-1;
  boss.fireCooldown-=dt;boss.flexCooldown-=dt;boss.speechCooldown-=dt;
  if(boss.speechCooldown<=0){boss.speechText=randomMrLeeShout();boss.speechUntil=time+1.5;boss.speechCooldown=rand(2.5,5.6)}
  if(boss.flexCooldown<=0){boss.flexUntil=time+.72;boss.flexCooldown=rand(2.4,4.4)}
  if(boss.fireCooldown<=0){const count=boss.shotsPerAttack;for(let i=0;i<count;i++){const spread=(i-(count-1)/2)*.17;throwFood(boss,chooseFood(),spread,boss.foodSpeed)}boss.fireCooldown=boss.fireRate}
  if(rects(player,boss)){
    const stomp=player.vy>0&&prevY+player.h<=boss.y+25;
    if(stomp){player.y=boss.y-player.h-3;player.vy=-520;boss.flexUntil=time+.7;bossHit(10);say(`💪 STOMP! ${boss.name.toUpperCase()} FLEXES!`,.5)}
    else death(`${boss.name} caught you!`,false);
  }
}

function update(dt){
  time+=dt;if(messageTimer>0){messageTimer-=dt;if(messageTimer<=0)messageEl.style.opacity="0"}
  const prevY=player.y;let dir=0;if(keys.ArrowLeft||keys.KeyA)dir--;if(keys.ArrowRight||keys.KeyD)dir++;
  const enamSpeed=handsOn()?1.65:1;
  const speed=(325+Math.min(level*5,40))*(player.ridingRary?1.28:1)*enamSpeed,accel=(player.ridingRary?3150:2500)*(handsOn()?1.45:1);
  if(dir){player.vx+=dir*accel*dt;player.facing=dir}else player.vx*=Math.pow(.002,dt);
  player.vx=clamp(player.vx,-speed,speed);
  if(keys.Space){player.vy-=2450*dt;player.vy=Math.max(player.vy,-540)}
  player.vy+=GRAV*dt;player.x+=player.vx*dt;player.x=clamp(player.x,0,worldWidth-player.w);player.y+=player.vy*dt;resolvePlatforms(dt);
  if(player.y>H+180)death(`You fell off the course!`,true);

  for(const p of wordPickups)if(!p.collected&&rects(player,p))collectWord(p);
  if(expected<WORDS.length){
    for(const d of decoyPickups){
      if(!d.collected&&rects(player,d)){
        const needed=WORDS[expected]?.[0]||"the next number";
        death(`Wrong answer! You touched <b>${d.word.toUpperCase()}</b> but need <b>${needed.toUpperCase()}</b>.`,true);
        break;
      }
    }
  }
  for(const b of bonuses)if(!b.collected&&rects(player,b))collectBonus(b);
  if(raryPickup&&!raryPickup.collected&&!raryUsed&&rects(player,raryPickup))collectRary();
  if(keyPickup&&!keyPickup.collected&&rects(player,keyPickup))collectKey();

  if(handsOn()&&time>=enamTujuhBurstAt){
    for(let i=0;i<12;i++)shootOne(i*Math.PI*2/12);
    enamTujuhBurstAt=time+.42;
  }

  updateEnemies(dt,prevY);updateBoss(dt,prevY);updateBossPowerups();

  for(const s of shots){
    s.x+=s.vx*dt;s.y+=s.vy*dt;s.life-=dt;const sb={x:s.x-s.r,y:s.y-s.r,w:s.r*2,h:s.r*2};
    if(boss&&boss.alive&&rects(sb,boss)){s.life=0;bossHit(1);continue}
    for(const e of enemies){if(!e.alive||e.state==="bin"||e.state==="deadbin"||e.state==="broccoli")continue;if(rects(sb,e)){e.state="deadbin";e.stateUntil=Infinity;s.life=0;particlesAt(e.x+e.w/2,e.y+e.h/2,30,"#aeb8c2");say(`🏐 dodgeball! MR LEE IS NOW A BIN!`,.72);break}}
  }
  shots=shots.filter(s=>s.life>0&&s.x>-100&&s.x<worldWidth+100&&s.y>-100&&s.y<H+120);

  for(const f of foods){f.x+=f.vx*dt;f.y+=f.vy*dt;f.vy+=260*dt;f.life-=dt;const fb={x:f.x-f.r,y:f.y-f.r,w:f.r*2,h:f.r*2};if(rects(fb,player)){f.life=0;hitByFood(f)}}
  foods=foods.filter(f=>f.life>0&&f.y<H+130);
  for(const p of particles){p.x+=p.vx*dt;p.y+=p.vy*dt;p.vy+=480*dt;p.life-=dt}
  particles=particles.filter(p=>p.life>0);
  if(hasKey&&exitDoor.open&&rects(player,exitDoor))completeLevel();
  cameraX+=(clamp(player.x-W*.34,0,worldWidth-W)-cameraX)*Math.min(1,dt*6);updateHUD();
}
