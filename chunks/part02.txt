function particlesAt(x,y,count=14,color="#ffd84d"){
  for(let i=0;i<count;i++){const a=rand(0,Math.PI*2),s=rand(50,240);particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,life:rand(.3,.8),size:rand(3,8),color})}
}

function death(reason,force=true){
  if(levelTransition)return;
  if(!force&&(achOn()||time<player.ignoreHitsUntil))return;
  if(!force&&player.ridingRary){
    player.ridingRary=false;raryUsed=true;player.ignoreHitsUntil=time+1.4;
    say(`🦁 RARY SAVED YOU!<br><span style="font-size:.55em">RARY runs away. MAX AURA!</span>`,1.6);
    particlesAt(player.x,player.y,28,"#f5c44d");return;
  }
  deaths++;running=false;levelTransition=true;
  say(`${reason}<br><span style="font-size:.72em">Sampai Jumpa... Mr Lee wins!</span>`,1.75);

  const restartLevel=level;
  const useBossCheckpoint=level===10&&hasKey&&(bossIndex>=0||bossCheckpointMode);
  const resumeBossIndex=bossCheckpointMode?bossCheckpointIndex:Math.max(0,bossIndex);

  setTimeout(()=>{
    if(!levelTransition)return;
    if(useBossCheckpoint){
      initLevel10BossCheckpoint(resumeBossIndex);
    }else{
      initLevel(restartLevel);
    }
    running=true;
    last=performance.now();
    requestAnimationFrame(loop);
  },1450);
}

function collectWord(p){
  if(p.collected)return;

  if(p.num!==expected){
    const needed=expected<=10?WORDS[expected][0].toUpperCase():"THE KEY";
    death(`Wrong number! You collected <b>${p.word.toUpperCase()}</b> instead of <b>${needed}</b>.`,true);
    return;
  }

  p.collected=true;
  expected++;
  particlesAt(p.x+p.w/2,p.y+p.h/2,20);

  // Every correct pickup explicitly reinforces the Indonesian number.
  speakIndonesianNumber(p.word);

  if(p.num===7){
    player.enamTujuhUntil=time+20;
    enamTujuhBurstAt=0;
    say(`<b>${p.word.toUpperCase()}!</b><br><span style="font-size:.56em">🙌⚡ ENAM TUJUH! 20 SEC SUPER SPEED + AUTOMATIC 360° DODGEBALL FIRE!</span>`,2);
  }else if(level===1&&p.num===0){
    say(`<b>NOL!</b><br><span style="font-size:.55em">GOOD! NEXT IS SATU →</span>`,1.3);
  }else if(level===1&&p.num===2){
    say(`<b>DUA!</b><br><span style="font-size:.55em">NOW TRY FLYING HIGHER WITH SPACE.</span>`,1.35);
  }else if(level===1&&p.num===4){
    say(`<b>EMPAT!</b><br><span style="font-size:.55em">PRESS F TO THROW A PINK DODGEBALL AT MINI MR LEE.</span>`,1.5);
  }else{
    say(`<b>${p.word.toUpperCase()}!</b><br><span style="font-size:.55em">${p.num}</span>`,.8);
  }

  if(expected===11){
    if(level===10&&bossCheckpointMode){
      bossCheckpointMode=false;
      say(`<b>SEPULUH!</b><br><span style="font-size:.55em">MR LEE IS BACK — FIGHT!</span>`,1.2);
      setTimeout(()=>{if(running&&!levelTransition)startBossSequence(bossCheckpointIndex)},650);
    }else{
      spawnKey();
    }
  }
}

function spawnKey(){
  keySpawned=true;
  // The number quiz is finished. Remove wrong-answer cards so they cannot
  // trigger an invalid "next number" after SEPULUH while the player hunts for the key.
  decoyPickups=[];
  const occupied=[
    ...bonuses.filter(b=>!b.collected),
    ...(raryPickup&&!raryPickup.collected&&!raryUsed?[raryPickup]:[])
  ];
  const spots=safeSpots(80);
  let placed=null;
  for(const spot of spots){
    const candidate={x:spot.x-24,y:spot.y-4,w:48,h:38};
    if(candidate.x<280||candidate.x>worldWidth-440)continue;
    if(!overlapsAny(candidate,occupied,45)){placed=candidate;break}
  }
  if(!placed)placed={x:worldWidth*.52,y:320,w:48,h:38};
  keyPickup={...placed,collected:false};
  say(`🔑 ALL NUMBERS COMPLETE!<br><span style="font-size:.55em">THE KEY HAS APPEARED SOMEWHERE IN THE LEVEL!</span>`,1.9);
}
